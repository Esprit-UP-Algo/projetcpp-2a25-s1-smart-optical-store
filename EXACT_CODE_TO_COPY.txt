void FournisseurWindow::configureRoleBasedAccess()
{
    qDebug() << "Configuring access for role:" << userRole;
    
    // If user is fournisseur role, hide navigation to other modules
    if (userRole.toLower() == "fournisseur")
    {
        // Hide navigation buttons to other modules
        ui->pushButton->setVisible(false);      // Stock button
        ui->pushButton_2->setVisible(false);    // Ventes button
        ui->pushButton_3->setVisible(false);    // Clients button
        ui->pushButton_4->setVisible(false);    // Employés button
        // pushButton_5 is Fournisseur - keep it visible (don't hide it)
        
        // Optionally hide the entire navigation panel
        // ui->groupBox->setVisible(false);  // Uncomment this to hide entire "Tableau de bord" group
        
        qDebug() << "Role-based access configured: fournisseur role - navigation restricted";
        
        // Show info message (optional - remove if you don't want the popup)
        QMessageBox::information(this, "Mode Fournisseur",
            "Vous êtes connecté en mode Fournisseur.\n"
            "Accès limité au module de gestion des fournisseurs.");
    }
    else if (userRole.toLower() == "admin" || userRole.isEmpty())
    {
        // Full access for admin - all buttons visible (default state)
        qDebug() << "Full access granted for role:" << userRole;
    }
    else
    {
        // For other roles, you can add specific restrictions here
        qDebug() << "Unknown role:" << userRole << "- defaulting to full access";
    }
}
